# Crawling is deliberately ALLOWED here.
#
# De-indexing, when wanted, belongs in a host-level X-Robots-Tag: noindex
# header (GitHub Pages sends none). A crawler has to be able to fetch a page
# to see such a header — a blanket "Disallow: /" would block the fetch, so a
# noindex would never be read and already-indexed pages would linger.
#
# Do not add a blanket Disallow unless the site is already out of the index.

User-agent: *
Allow: /
